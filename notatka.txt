2. wyświetl uczniów którzy są z łodzi i ich hobby to programowanie

4. wyświetl uczniów z trzech wskazanych miejscowości
SELECT * FROM `uczniowie` where miejscowosc in( "Łódź", "Warszawa", "Kraków")order by miejscowosc ASC ;
5. wyświetl uczniów, którzy nie mieszkają w warszawie i krakowie
SELECT imie, nazwisko, miejscowosc FROM `uczniowie` where miejscowosc NOT in( "Warszawa", "Kraków");

9. policz wszystkich uczniów z łodzi

10. Wyświetl hobby, które występują u co najmniej 10 uczniów
SELECT hobby, COUNT(*) AS liczba_uczniów FROM uczniowie group by hobby having count(*) >= 10 order by liczba_uczniów desc;

23.1 Wyświetl uczniów urodzonych w czerwcu i grudniu
Select imie, nazwisko, data_urodzenia from uczniowie where MONTH(data_urodzenia) IN (6,12) ORDER BY data_urodzenia;

23.2 Wyświetl uczniów urodzonych w czerwcu lub grudniu
Select imie, nazwisko, data_urodzenia from uczniowie where MONTH(data_urodzenia) = 6 OR MONTH(data_urodzenia) = 12 ORDER BY data_urodzenia;

24. oblicz wiek kazdego ucznia 
SELECT imie, nazwisko, data_urodzenia, TIMESTAMPDIFF( YEAR, data_urodzenia, CURDATE() ) AS wiek from uczniowie order by wiek desc;

25. Wyświetl uczniów, których adres e-mail zawiera nazwisko "kowalski"
SELECT imie, nazwisko, email from uczniowie where email like '%kowalski%';
28. Wyświetlanie najstarszego i najmłodszego ucznia według daty urodzenia (same daty )
Select MIN(data_urodzenia) AS najstarszy, MAX(data_urodzenia) AS najmlodszy,from uczniowie;
29. Wyświetl uczniów mieszkających w najliczniej reprezentowanej miejscowści
Select imie, nazwisko, miejscowosc from uczniowie where miejscowosc = (select miejscowos.c from uczniowie group by miejscowosc order by count(*) DESC, miejscowosc ASC limit 1) order by nazwisko, imie;
31. Wyświetl uczniów, których imie ma dokładnie 5 liter
Select id, imie, nazwisko FROM uczniowie where CHAR_LENGTH(imie) = 5 order by imie;
